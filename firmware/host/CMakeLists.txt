cmake_minimum_required(VERSION 3.20)
project(carbentra_switch_host C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(FIRMWARE "${CMAKE_CURRENT_LIST_DIR}/..")
if(DEFINED ENV{CJSON_ROOT})
  set(CJSON "$ENV{CJSON_ROOT}")
elseif(DEFINED ENV{IDF_PATH})
  set(CJSON "$ENV{IDF_PATH}/components/json/cJSON")
else()
  message(FATAL_ERROR "Set CJSON_ROOT to the pinned cJSON checkout, or IDF_PATH to ESP-IDF 5.4.3")
endif()
# Native environment paths can contain CMake escapes such as \\a on Windows.
# Normalize at the consumer so both the coordinator and direct CMake calls work.
file(TO_CMAKE_PATH "${CJSON}" CJSON)
if(NOT EXISTS "${CJSON}/cJSON.c")
  message(FATAL_ERROR "The cJSON source checkout is incomplete: ${CJSON}")
endif()
add_executable(test_core "${FIRMWARE}/core/switch_core.c" "${FIRMWARE}/core/meter.c"
  "${FIRMWARE}/main/command_json.c" "${FIRMWARE}/tests/test_core.c" "${CJSON}/cJSON.c")
target_include_directories(test_core PRIVATE "${FIRMWARE}/core" "${FIRMWARE}/main" "${CJSON}")
if(MSVC)
  target_compile_options(test_core PRIVATE /W4 /WX /wd4996)
else()
  target_compile_options(test_core PRIVATE -Wall -Wextra -Werror -Wpedantic)
  target_link_libraries(test_core PRIVATE m)
endif()
enable_testing()
add_test(NAME switch_control_meter_protocol COMMAND test_core)
